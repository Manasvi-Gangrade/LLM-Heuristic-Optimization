Champion Heuristic (GPT-OSS 120B)
import random

def heuristic(problem_instance):
    n = problem_instance["n_cities"]
    dist = problem_instance["dist_matrix"]
    if n <= 1:
        return {"tour": list(range(n)), "cost": 0.0}
    samples = max(10, n // 5)
    best_tour = None
    best_cost = float('inf')
    base = list(range(n))
    for _ in range(samples):
        tour = base[:]
        random.shuffle(tour)
        improved = True
        while improved:
            improved = False
            for i in range(1, n - 1):
                a = tour[i - 1]
                b = tour[i]
                for k in range(i + 1, n):
                    c = tour[k]
                    d = tour[(k + 1) % n]
                    delta = dist[a][c] + dist[b][d] - dist[a][b] - dist[c][d]
                    if delta < -1e-12:
                        tour[i:k + 1] = reversed(tour[i:k + 1])
                        improved = True
        cost = sum(dist[tour[i]][tour[(i + 1) % n]] for i in range(n))
        if cost < best_cost:
            best_cost = cost
            best_tour = tour[:]
    if best_tour is None:
        tour = [0]
        unvisited = set(range(1, n))
        while unvisited:
            cur = tour[-1]
            nxt = min(unvisited, key=lambda c: dist[cur][c])
            tour.append(nxt)
            unvisited.remove(nxt)
        best_tour = tour
        best_cost = sum(dist[tour[i]][tour[(i + 1) % n]] for i in range(n))
    return {"tour": best_tour, "cost": float(best_cost)}
